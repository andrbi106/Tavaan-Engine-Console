## استفاده

```php
use Tavaan\Engine\Console\Command;
use Tavaan\Engine\Console\Application;
use Tavaan\Engine\Console\ConsoleServiceProvider;
use Tavaan\Engine\Console\ArgvParser;
use Tavaan\Engine\Console\SignatureParser;

// تعریف Command
final class GreetCommand extends Command
{
    protected const NAME = 'greet';
    protected const DESCRIPTION = 'Greet someone';

    public function handle(): int
    {
        $name = $this->argument(0, 'World');
        $this->info("Hello, {$name}!");
        return self::SUCCESS;
    }
}

// Setup
$provider = new ConsoleServiceProvider([]);
$provider->register();
$app = $provider->getApplication();
$app->add(new GreetCommand());

// اجرا
exit($app->run($argv));

// پارس مستقیم argv
[$args, $opts] = ArgvParser::parse(['--name=ali', '--verbose', 'value']);

// پارس signature
$signature = SignatureParser::parse('cache:clear {store? : نام store} {--force}');
// ['name' => 'cache:clear', 'arguments' => [...], 'options' => [...]]
```
